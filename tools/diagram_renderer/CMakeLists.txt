cmake_minimum_required(VERSION 3.21)
project(DiagramRenderer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Gui Svg)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Gui Svg)
add_executable(RenderDiagrams render.cpp)
target_link_libraries(RenderDiagrams PRIVATE Qt${QT_VERSION_MAJOR}::Gui Qt${QT_VERSION_MAJOR}::Svg)
